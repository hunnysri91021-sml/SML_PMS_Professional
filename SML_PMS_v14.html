2. Login + Token
async function getGraphToken() {
  const accounts = msalInstance.getAllAccounts();
  if (accounts.length === 0) {
    await msalInstance.loginPopup({ scopes: GRAPH_SCOPES });
  }
  const response = await msalInstance.acquireTokenSilent({
    scopes:  GRAPH_SCOPES,
    account: msalInstance.getAllAccounts()[0]
  });
  return response.accessToken;
}
3. อ่านพนักงานจาก Excel (Sheet "Employees")
async function readEmployeesFromExcel(token, siteId, fileId) {
  const res = await fetch(
    `https://graph.microsoft.com/v1.0/sites/${siteId}`
    + `/drive/items/${fileId}/workbook/worksheets/Employees/usedRange`,
    { headers: { Authorization: `Bearer ${token}` } }
  );
  const { values } = await res.json();      // 2D array [row][col]
  return values.slice(1)                    // ข้าม header
    .filter(r => r[0])                      // กรองแถวว่าง
    .map(r => ({
      id:r[0], name:r[1], role:r[2], dept:r[3],
      position:r[4], grade:r[5], l1:r[6], l2:r[7],
      exec:r[8], status:r[9], lastLogin:r[10]
    }));
}
4. บันทึกผลประเมินลง Excel (Sheet "Appraisals")
async function saveAppraisalToExcel(token, siteId, fileId, data) {
  // หาแถวถัดไป
  const used = await (await fetch(
    `https://graph.microsoft.com/v1.0/sites/${siteId}`
    + `/drive/items/${fileId}/workbook/worksheets/Appraisals`
    + `/usedRange(valuesOnly=true)`,
    { headers: { Authorization: `Bearer ${token}` } }
  )).json();
  const nextRow = (used?.rowCount ?? 1) + 1;

  // เขียนแถวใหม่
  await fetch(
    `https://graph.microsoft.com/v1.0/sites/${siteId}`
    + `/drive/items/${fileId}/workbook/worksheets/Appraisals`
    + `/range(address='A${nextRow}')`,
    {
      method:  'PATCH',
      headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' },
      body:    JSON.stringify({ values: [[
        data.empId, data.cycle, data.level,
        ...data.scores,   // m0–m13
        data.Y, data.Z, data.X, data.grade,
        'Submitted', new Date().toISOString()
      ]] })
    }
  );
}
