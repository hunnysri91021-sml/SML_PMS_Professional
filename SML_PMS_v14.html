  3. แนะนำให้ IT จำกัดสิทธิ์แอปนี้ด้วย SharePoint Application Access Policy ให้เข้าถึงได้แค่ไซต์ Chosiya-HR ไซต์เดียว ไม่ใช่ทั้งองค์กร (ลดความเสี่ยงถ้ากุญแจหลุด)

  5. เอา URL ของ Worker (เช่น https://sml-pms-proxy.xxx.workers.dev) กับ PROXY_API_KEY มากรอกช่องด้านล่าง แล้วกดบันทึก

async function getGraphToken() {
  const accounts = msalInstance.getAllAccounts();
  if (accounts.length === 0) {
    await msalInstance.loginPopup({ scopes: GRAPH_SCOPES });
  }
  const response = await msalInstance.acquireTokenSilent({
    scopes:  GRAPH_SCOPES,
    account: msalInstance.getAllAccounts()[0]
  });
  return response.accessToken;
}

async function readEmployeesFromExcel(token, siteId, fileId) {
  const res = await fetch(
    `https://graph.microsoft.com/v1.0/sites/${siteId}`
    + `/drive/items/${fileId}/workbook/worksheets/Employees/usedRange`,
    { headers: { Authorization: `Bearer ${token}` } }
  );
  const { values } = await res.json();      // 2D array [row][col]
  return values.slice(1)                    // ข้าม header
    .filter(r => r[0])                      // กรองแถวว่าง
    .map(r => ({
      id:r[0], name:r[1], role:r[2], dept:r[3],
      position:r[4], grade:r[5], l1:r[6], l2:r[7],
      exec:r[8], status:r[9], lastLogin:r[10]
    }));
}
